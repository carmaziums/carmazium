  <action>
Create `mobile/app/notification-prefs/index.tsx`.

The screen has two sections:
1. **Category toggles** — four rows (Bids, Offers, Messages, System) each with a label, description, and a Switch
2. **Quiet hours** — an enable/disable toggle and two time inputs (Start, End) that appear when quiet hours is on

Load preferences from `getNotificationPreferences()` in a `useEffect` on mount. Save on every toggle change using `saveNotificationPreferences()`.

For time selection: Use a plain text input styled to look like a picker. Quiet hours time is stored as a string `"HH:MM"`. On press, show an Alert or inline text input that allows the user to type the time. This avoids requiring `@react-native-community/datetimepicker` which would need a dev build plugin config. A simple inline approach is sufficient for Phase 1.

```typescript
import React, { useEffect, useState } from 'react';
import {
    View,
    Text,
    Switch,
    ScrollView,
    TouchableOpacity,
    TextInput,
    Alert,
    ActivityIndicator,
} from 'react-native';
import { useRouter } from 'expo-router';
import {
    NotificationPreferences,
    DEFAULT_PREFS,
    getNotificationPreferences,
    saveNotificationPreferences,
} from '@/lib/notifications';

// ---- Toggle Row ----

interface ToggleRowProps {
    label: string;
    description: string;
    value: boolean;
    onToggle: (val: boolean) => void;
}

function ToggleRow({ label, description, value, onToggle }: ToggleRowProps) {
    return (
        <View className="flex-row items-center justify-between px-4 py-4 border-b border-white/5">
            <View className="flex-1 pr-4">
                <Text className="text-white font-semibold text-sm">{label}</Text>
                <Text className="text-white/50 text-xs mt-0.5">{description}</Text>
            </View>
            <Switch
                value={value}
                onValueChange={onToggle}
                trackColor={{ false: '#2a2d36', true: '#ff0037' }}
                thumbColor={value ? '#ffffff' : '#6b7280'}
                ios_backgroundColor="#2a2d36"
            />
        </View>
    );
}

// ---- Time Row ----

interface TimeRowProps {
    label: string;
    value: string;          // "HH:MM"
    onPress: () => void;
    disabled?: boolean;
}

function TimeRow({ label, value, onPress, disabled }: TimeRowProps) {
    return (
        <TouchableOpacity
            onPress={onPress}
            disabled={disabled}
            className={`flex-row items-center justify-between px-4 py-4 border-b border-white/5 ${disabled ? 'opacity-40' : ''}`}
            activeOpacity={0.7}
        >
            <Text className="text-white/70 text-sm">{label}</Text>
            <View className="flex-row items-center gap-2">
                <Text className="text-white font-medium">{value}</Text>
                <Text className="text-white/30">›</Text>
            </View>
        </TouchableOpacity>
    );
}

// ---- Screen ----

export default function NotificationPrefsScreen() {
    const router = useRouter();
    const [prefs, setPrefs] = useState<NotificationPreferences>(DEFAULT_PREFS);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        getNotificationPreferences().then((p) => {
            setPrefs(p);
            setLoading(false);
        });
    }, []);

    const handleToggle = async (key: keyof NotificationPreferences, value: boolean | string) => {
        const updated = { ...prefs, [key]: value };
        setPrefs(updated);
        await saveNotificationPreferences({ [key]: value });
    };

    const handleTimePress = (field: 'quietStart' | 'quietEnd') => {
        const current = prefs[field];
        Alert.prompt(
            field === 'quietStart' ? 'Quiet hours start' : 'Quiet hours end',
            'Enter time in HH:MM format (24-hour)',
            [
                { text: 'Cancel', style: 'cancel' },
                {
                    text: 'Save',
                    onPress: (text) => {
                        if (!text) return;
                        // Validate HH:MM format
                        const match = text.match(/^([01]\d|2[0-3]):([0-5]\d)$/);
                        if (!match) {
                            Alert.alert('Invalid time', 'Please enter time as HH:MM (e.g. 22:00)');
                            return;
                        }
                        handleToggle(field, text);
                    },
                },
            ],
            'plain-text',
            current,
        );
    };

    if (loading) {
        return (
            <View className="flex-1 bg-[#0a0d14] items-center justify-center">
                <ActivityIndicator color="#ff0037" />
            </View>
        );
    }

    return (
        <View className="flex-1 bg-[#0a0d14]">
            {/* Header */}
            <View className="px-4 pt-14 pb-4 flex-row items-center gap-3">
                <TouchableOpacity onPress={() => router.back()} className="p-2 -ml-2">
                    <Text className="text-white/60 text-base">←</Text>
                </TouchableOpacity>
                <Text className="text-white font-bold text-lg">Notification Preferences</Text>
            </View>

            <ScrollView showsVerticalScrollIndicator={false}>
                {/* Category Toggles */}
                <View className="mt-2">
                    <Text className="text-white/40 text-xs font-semibold uppercase tracking-widest px-4 pb-2">
                        Notification Types
                    </Text>
                    <View className="bg-white/[0.03] rounded-xl mx-4 overflow-hidden">
                        <ToggleRow
                            label="Auction Bids"
                            description="Outbid alerts and auction results"
                            value={prefs.bids}
                            onToggle={(v) => handleToggle('bids', v)}
                        />
                        <ToggleRow
                            label="Offers"
                            description="New offers, counter-offers, and acceptances"
                            value={prefs.offers}
                            onToggle={(v) => handleToggle('offers', v)}
                        />
                        <ToggleRow
                            label="Messages"
                            description="New chat messages from buyers and sellers"
                            value={prefs.messages}
                            onToggle={(v) => handleToggle('messages', v)}
                        />
                        <ToggleRow
                            label="System"
                            description="KYC updates, account alerts, and announcements"
                            value={prefs.system}
                            onToggle={(v) => handleToggle('system', v)}
                        />
                    </View>
                </View>

                {/* Quiet Hours */}
                <View className="mt-6 mb-10">
                    <Text className="text-white/40 text-xs font-semibold uppercase tracking-widest px-4 pb-2">
                        Quiet Hours
                    </Text>
                    <View className="bg-white/[0.03] rounded-xl mx-4 overflow-hidden">
                        <ToggleRow
                            label="Enable Quiet Hours"
                            description="Suppress local notification banners during set hours"
                            value={prefs.quietHoursEnabled}
                            onToggle={(v) => handleToggle('quietHoursEnabled', v)}
                        />
                        <TimeRow
                            label="Start time"
                            value={prefs.quietStart}
                            onPress={() => handleTimePress('quietStart')}
                            disabled={!prefs.quietHoursEnabled}
                        />
                        <TimeRow
                            label="End time"
                            value={prefs.quietEnd}
                            onPress={() => handleTimePress('quietEnd')}
                            disabled={!prefs.quietHoursEnabled}
                        />
                    </View>
                    <Text className="text-white/30 text-xs px-4 mt-3">
                        Note: Quiet hours suppresses local banners only. Push notifications may still appear if your device is in the background.
                    </Text>
                </View>
            </ScrollView>
        </View>
    );
}
```

`Alert.prompt` is iOS-only. For Android, replace with an inline `TextInput` approach:

```typescript
// After the TimeRow components — add an inline input that appears when editing:
// Check Platform.OS before using Alert.prompt
import { Platform } from 'react-native';

const handleTimePress = (field: 'quietStart' | 'quietEnd') => {
    if (Platform.OS === 'ios') {
        Alert.prompt(/* ... as above */);
    } else {
        // Android: show an Alert with a TextInput-based dialog
        // Use a simple two-button Alert on Android with instruction text
        // The user can still edit by using a custom modal or inline state toggle
        // For Phase 1, show the current value and a brief instruction
        Alert.alert(
            field === 'quietStart' ? 'Quiet hours start' : 'Quiet hours end',
            `Current: ${prefs[field]}\n\nEdit quiet hours time in Settings (this picker requires a native modal — will be improved in Phase 8).`,
            [{ text: 'OK' }],
        );
    }
};
```

This is an acceptable Phase 1 limitation for Android quiet-hours time editing. The toggle enable/disable works on all platforms. Full time picker UI is deferred to Phase 8 (app store prep phase includes UI polish).
