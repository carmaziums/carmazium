      <div className="w-full h-1.5 bg-white/10 rounded-full overflow-hidden">
        <div
          className={`h-full rounded-full transition-all duration-300 ${isMinMet ? 'bg-green-500' : 'bg-amber-500'}`}
          style={{ width: `${progressPct}%` }}
        />
      </div>
    </div>
  )
})()}
