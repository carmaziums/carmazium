    ```tsx
    </div>
    ```

    Radius check logic (uses LocationContext buyer GPS postcode — this is the sidebar hint, NOT a gate):
    - Import `useLocationContext` (from src/context/LocationContext.tsx)
    - Get buyer's postcode from context: `const { location } = useLocationContext()`
    - Only perform the check if `listing.deliveryMaxMiles` is set AND buyer GPS location is available
    - Use the /api/delivery-distance route to get distanceMiles (one-time fetch on mount, not on every render):
      ```typescript
      const [deliveryDistanceInfo, setDeliveryDistanceInfo] = React.useState<{ distanceMiles: number } | null>(null);

      React.useEffect(() => {
        if (!listing.deliveryAvailable || !listing.deliveryMaxMiles || !listing.latitude || !listing.longitude || !location?.postcode) return;
        fetch(`/api/delivery-distance?originLat=${listing.latitude}&originLng=${listing.longitude}&postcode=${location.postcode}&pricePerMile=0`)
          .then(r => r.ok ? r.json() : null)
          .then(data => data && setDeliveryDistanceInfo(data))
          .catch(() => {});
      }, [listing.id, location?.postcode]);

      const isOutsideRadius = listing.deliveryMaxMiles && deliveryDistanceInfo
        ? deliveryDistanceInfo.distanceMiles > listing.deliveryMaxMiles
        : false;
      ```

    Conditional render:
    ```tsx
    {listing.deliveryAvailable && (
      isOutsideRadius
        ? <OutsideRadiusCard maxMiles={listing.deliveryMaxMiles} />
        : <DeliveryAvailableCard listing={listing} />
    )}
    ```
